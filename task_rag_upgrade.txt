app/rag/store.py (CREATE)
============================
Module-level singletons so the engine and store are reused across requests.

    from typing import Optional
    from langchain_openai import OpenAIEmbeddings
    from langchain_postgres import PGEngine, PGVectorStore
    from app.config import settings

    TABLE = "meeting_embeddings"
    VECTOR_SIZE = 1536   # text-embedding-3-small

    _engine: Optional[PGEngine] = None
    _store: Optional[PGVectorStore] = None
    _embeddings: Optional[OpenAIEmbeddings] = None

    def get_engine() -> PGEngine:
        global _engine
        if _engine is None:
            _engine = PGEngine.from_connection_string(url=settings.vector_db_url)
        return _engine

    def get_embeddings() -> OpenAIEmbeddings:
        global _embeddings
        if _embeddings is None:
            _embeddings = OpenAIEmbeddings(model="text-embedding-3-small")
        return _embeddings

    async def init_table() -> None:
        """Create the vector table if missing. Idempotent."""
        await get_engine().ainit_vectorstore_table(
            table_name=TABLE,
            vector_size=VECTOR_SIZE,
        )

    def get_store() -> PGVectorStore:
        global _store
        if _store is None:
            _store = PGVectorStore.create_sync(
                engine=get_engine(),
                table_name=TABLE,
                embedding_service=get_embeddings(),
            )
        return _store

app/rag/indexer.py (CREATE)
============================
Pulls meetings from meeting-processor and indexes them into the vector store.
Two responsibilities:
  1. index_meeting(meeting_dict) — embed a single meeting
  2. sync_all() — pull all meetings from processor and index any that are missing

Design decision: build a single document per meeting combining title, summary,
action items, decisions, and risks into one page_content string. Semantic
search works better on coherent chunks than on fragmented per-section docs.

Also maintain an in-process set of indexed meeting IDs so re-indexing the
same meeting doesn't duplicate documents. On first call to sync_all in a
fresh process, this set is empty — so it will re-index everything. That's
acceptable for a portfolio project; note it in a comment.

    import logging
    import httpx
    from langchain_core.documents import Document
    from app.config import settings
    from app.rag.store import get_store

    log = logging.getLogger(__name__)
    _indexed_ids: set[str] = set()

    def _meeting_to_document(meeting: dict) -> Document:
        """Combine meeting fields into one searchable text blob."""
        title = meeting.get("title", "")
        analysis = meeting.get("analysis") or {}
        parts = [f"Meeting: {title}"]
        if analysis.get("summary"):
            parts.append(f"Summary: {analysis['summary']}")
        for ai in analysis.get("action_items", []) or []:
            parts.append(
                f"Action item: {ai.get('task')} assigned to {ai.get('owner')}"
                + (f" due {ai['due_date']}" if ai.get("due_date") else "")
            )
        for d in analysis.get("decisions", []) or []:
            parts.append(f"Decision: {d.get('description')}")
        for r in analysis.get("risks", []) or []:
            parts.append(f"Risk: {r.get('description')}")
        return Document(
            page_content="\n".join(parts),
            metadata={
                "meeting_id": str(meeting.get("id")),
                "title": title,
            },
        )

    def index_meeting(meeting: dict) -> None:
        """Embed one meeting. Skips if already indexed in this process."""
        mid = str(meeting.get("id"))
        if mid in _indexed_ids:
            return
        doc = _meeting_to_document(meeting)
        get_store().add_documents([doc])
        _indexed_ids.add(mid)
        log.info("Indexed meeting %s", mid)

    async def sync_all() -> dict:
        """Pull all meetings from processor and index any missing."""
        async with httpx.AsyncClient() as client:
            r = await client.get(
                f"{settings.meeting_processor_url}/meetings", timeout=10.0
            )
            r.raise_for_status()
            summaries = r.json()

        indexed = 0
        skipped = 0
        failed = 0
        for s in summaries:
            mid = str(s.get("id"))
            if mid in _indexed_ids:
                skipped += 1
                continue
            try:
                async with httpx.AsyncClient() as client:
                    rd = await client.get(
                        f"{settings.meeting_processor_url}/meetings/{mid}",
                        timeout=10.0,
                    )
                    rd.raise_for_status()
                    full = rd.json()
                index_meeting(full)
                indexed += 1
            except Exception as e:
                log.warning("Failed to index meeting %s: %s", mid, e)
                failed += 1

        return {"indexed": indexed, "skipped": skipped, "failed": failed, "total": len(summaries)}

    async def search_meetings(query: str, k: int = 3) -> list[dict]:
        """Search past meetings by meaning using semantic similarity.

        Use this when the user refers to a meeting by topic, decision, or
        concept rather than an exact title or date. Returns the top k most
        semantically similar meetings, each with a snippet and meeting_id.

        Examples of good queries:
            "what did we decide about ad spend"
            "who is responsible for the hiring plan"
            "any risks around revenue"
        """
        from app.rag.store import get_store
        # PGVectorStore has an async variant; use it because ADK tools are async.
        # Check the installed class with `help(store.asimilarity_search)` first.
        store = get_store()
        try:
            results = await store.asimilarity_search(query, k=k)
        except AttributeError:
            # Fall back to sync if asimilarity_search isn't available in this version
            results = store.similarity_search(query, k=k)
        return [
            {
                "meeting_id": r.metadata.get("meeting_id"),
                "title": r.metadata.get("title"),
                "snippet": r.page_content[:400],
            }
            for r in results
        ]

    from contextlib import asynccontextmanager
    from fastapi import FastAPI
    from app.rag.store import init_table
    from app.rag.indexer import sync_all
    from app.routes import chat, index

    @asynccontextmanager
    async def lifespan(app: FastAPI):
        try:
            await init_table()
            result = await sync_all()
            print(f"[startup] indexed {result['indexed']} meetings, "
                  f"skipped {result['skipped']}, failed {result['failed']}")
        except Exception as e:
            # Do not crash the app if the processor is down — log and continue.

        yield
